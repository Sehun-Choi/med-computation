# --- Road Data Files ---
raw.data = read.table(file=raw_file, header=TRUE, row.names=1, sep="\t", na.strings="NULL")
head(raw.data)

raw.data <- raw.data[, -(1:5)]
head(raw.data)

group <- data.frame(
    row.names = colnames(raw.data),
    condition = group_condition
)
head(group)


# --- DEG Analysis (NAc_Sus vs. NAc_CTRL) ---
# Initializing the DESeq2 Object
dds <- DESeqDataSetFromMatrix(countData = raw.data, colData = group, design = ~ condition)

# The Low-Count Filter
smallestGroupSize = 2
keep <- rowSums(counts(dds) >= 10) >= smallestGroupSize
dds <- dds[keep,]
